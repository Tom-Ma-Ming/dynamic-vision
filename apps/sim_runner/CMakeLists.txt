add_executable(sim_runner main.cpp)
target_include_directories(sim_runner PRIVATE ${PROJECT_SOURCE_DIR}/include)
target_link_libraries(sim_runner PRIVATE vision_sim ${OpenCV_LIBS})

endforeach()
