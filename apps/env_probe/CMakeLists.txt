add_executable(env_probe main.cpp)
if(VISION_WITH_TRT)
  find_package(CUDAToolkit REQUIRED)
  set(TensorRT_ROOT "$ENV{TensorRT_ROOT}" CACHE PATH "Installed TensorRT SDK")
  find_path(TRT_INCLUDE NvInfer.h PATHS "${TensorRT_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
  find_file(TRT_DLL NAMES nvinfer_11.dll nvinfer_10.dll nvinfer.dll PATHS "${TensorRT_ROOT}/lib" "${TensorRT_ROOT}/bin" NO_DEFAULT_PATH REQUIRED)
  target_include_directories(env_probe PRIVATE "${TRT_INCLUDE}")
  target_link_libraries(env_probe PRIVATE CUDA::cudart_static)
  target_compile_definitions(env_probe PRIVATE VISION_PROBE_TRT TRT_DLL_PATH=L"${TRT_DLL}")
endif()
if(MSVC)
  target_compile_options(env_probe PRIVATE /utf-8)
  set_property(TARGET env_probe PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
