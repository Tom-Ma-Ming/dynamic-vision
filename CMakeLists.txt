cmake_minimum_required(VERSION 3.20)
project(dynamic_vision LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(VISION_WITH_OPENCV "启用 OpenCV（视频读取与叠加）" OFF)
option(VISION_WITH_TRT "启用 TensorRT 后端（仅 Windows）" OFF)
option(VISION_BUILD_TESTS "构建单元测试" ON)
find_package(Threads REQUIRED)

if(VISION_WITH_TRT AND NOT WIN32)
  message(FATAL_ERROR "VISION_WITH_TRT 只允许在 Windows 上开启")
endif()

option(VISION_PROBE_ONLY "Skip legacy dependency smoke library" OFF)
if(VISION_BUILD_TESTS)
  enable_testing()
endif()
add_subdirectory(apps/env_probe)
if(VISION_WITH_OPENCV)
  find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc imgcodecs videoio video)
  set(VISION_OPENCV_TARGETS ${OpenCV_LIBS})
  add_library(vision_mot STATIC src/tracking/tracker.cpp src/tracking/imm.cpp src/tracking/camera_motion.cpp)
  target_include_directories(vision_mot PUBLIC src/tracking)
  target_link_libraries(vision_mot PUBLIC ${OpenCV_LIBS})
  if(VISION_BUILD_TESTS)
    add_executable(mot_test tests/unit/test_tracker.cpp)
    target_link_libraries(mot_test PRIVATE vision_mot)
    add_executable(imm_test tests/unit/test_imm.cpp)
    target_link_libraries(imm_test PRIVATE vision_mot)
    add_test(NAME imm_numeric COMMAND imm_test ${PROJECT_SOURCE_DIR}/configs/tracker-imm.yml)
    add_test(NAME mot_imm COMMAND mot_test ${PROJECT_SOURCE_DIR}/configs/tracker-imm.yml)
    add_test(NAME mot_synthetic COMMAND mot_test ${PROJECT_SOURCE_DIR}/configs/tracker.yml)
    add_executable(camera_motion_test tests/unit/test_camera_motion.cpp)
    target_link_libraries(camera_motion_test PRIVATE vision_mot)
    add_test(NAME camera_motion COMMAND camera_motion_test ${PROJECT_SOURCE_DIR}/configs/tracker-camera.yml)
  endif()
endif()
if(VISION_WITH_TRT)
  add_subdirectory(apps/trt_infer)
endif()
if(VISION_BUILD_TESTS)
  add_test(NAME probe_version COMMAND env_probe --version)
  add_test(NAME probe_environment COMMAND env_probe)
  if(NOT VISION_WITH_TRT)
    set_tests_properties(probe_environment PROPERTIES WILL_FAIL TRUE)
  endif()
endif()
if(VISION_PROBE_ONLY)
  return()
endif()

include(cmake/Dependencies.cmake)

if(MSVC)
  add_compile_options(/utf-8)
endif()

# vision_core：纯 C++17，不依赖 CUDA、TensorRT、相机 SDK。
# 后续任务在 VISION_CORE_SOURCES 中追加源文件。
set(VISION_CORE_SOURCES src/core/Placeholder.cpp)
add_library(vision_core STATIC ${VISION_CORE_SOURCES})
target_include_directories(vision_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(vision_core PUBLIC
  yaml-cpp::yaml-cpp nlohmann_json::nlohmann_json picosha2 Threads::Threads)
if(VISION_WITH_OPENCV)
  target_link_libraries(vision_core PUBLIC ${VISION_OPENCV_TARGETS})
  target_compile_definitions(vision_core PUBLIC VISION_WITH_OPENCV=1)
endif()

if(VISION_WITH_OPENCV)
  add_library(vision_sim STATIC src/sim/simulator.cpp)
  target_include_directories(vision_sim PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
  target_link_libraries(vision_sim PUBLIC vision_mot ${OpenCV_LIBS})
  target_compile_options(vision_sim PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
  option(VISION_WITH_ONNX "Link ONNX Runtime and run a real YOLO graph" OFF)
  if(VISION_WITH_ONNX)
    if(NOT ONNXRUNTIME_ROOT)
      message(FATAL_ERROR "VISION_WITH_ONNX requires ONNXRUNTIME_ROOT")
    endif()
    add_library(vision_yolo STATIC src/detect/yolo_onnx.cpp)
    target_include_directories(vision_yolo PUBLIC
      ${CMAKE_CURRENT_SOURCE_DIR}/include
      ${CMAKE_CURRENT_SOURCE_DIR}/src/detect
      ${ONNXRUNTIME_ROOT}/include)
    target_link_libraries(vision_yolo PUBLIC ${ONNXRUNTIME_ROOT}/lib/libonnxruntime.so ${OpenCV_LIBS})
    set_target_properties(vision_yolo PROPERTIES
      BUILD_RPATH ${ONNXRUNTIME_ROOT}/lib)
  endif()
  add_subdirectory(apps/sim_runner)
  if(VISION_BUILD_TESTS)
    add_executable(sim_test tests/unit/test_sim.cpp)
    target_link_libraries(sim_test PRIVATE vision_sim)
    target_compile_options(sim_test PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
    add_test(NAME sim_closed_loop COMMAND sim_test ${PROJECT_SOURCE_DIR}/configs/tracker.yml)
  endif()
endif()

if(VISION_BUILD_TESTS)
  enable_testing()
  include(GoogleTest)
  # 后续任务在 VISION_TEST_SOURCES 中追加测试文件。
  set(VISION_TEST_SOURCES
    tests/unit/test_smoke.cpp
    tests/unit/test_queue.cpp
    tests/unit/test_frame_log.cpp
    tests/unit/test_stats.cpp)
  add_executable(vision_tests ${VISION_TEST_SOURCES})
  add_executable(control_test tests/unit/test_control.cpp)
  target_include_directories(control_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
  add_test(NAME control_contract COMMAND control_test)
  target_link_libraries(vision_tests PRIVATE vision_core GTest::gtest_main)
  gtest_discover_tests(vision_tests)
endif()
